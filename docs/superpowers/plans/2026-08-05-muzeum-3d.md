# Muzeum 3D — plan wdrożenia

> **Dla agentów:** WYMAGANY SUB-SKILL: użyj `superpowers:subagent-driven-development` (zalecane) albo `superpowers:executing-plans`, żeby wdrażać ten plan zadanie po zadaniu. Kroki mają składnię checkboxów (`- [ ]`) do śledzenia postępu.

**Cel:** Przebudować `museum.html` z wizualizacji na szynach w swobodny spacer 3D z prawdziwym oświetleniem, cieniami, post-processingiem i architekturą sześciu sal.

**Architektura:** Zachowujemy renderowanie sceny z `PROJECTS` i osiem autorskich eksponatów proceduralnych. Wymieniamy warstwę renderowania (cienie, ACES, mapa środowiskowa, bloom), zastępujemy otwarty korytarz budynkiem (atrium + amfilada), a nawigację scrollem — swobodnym chodzeniem z kolizjami. Monolit `js/museum.js` rozbijamy na sześć modułów ES.

**Stack:** three.js r169 przez import map (jsDelivr), addony `postprocessing/*`, `environments/RoomEnvironment`, `controls/PointerLockControls`, `math/Octree`, `math/Capsule`. Tekstury PBR z Poly Haven (CC0). Bez frameworków, bez kroku budowania.

**Spec:** `docs/superpowers/specs/2026-08-05-muzeum-3d-design.md`

## Ograniczenia globalne

Obowiązują w każdym zadaniu:

- **Wersja three.js: dokładnie `0.169.0`** we wszystkich URL-ach — mieszanie wersji między `three` a `three/addons/` wywala instanceof przy klasach geometrii.
- **Zero zależności npm, zero kroku budowania.** Repo to czysty statyczny hosting na GitHub Pages.
- **Sufit wagi assetów: 10 MB.** Cel realny: ~4,5 MB. Po każdym dodaniu plików sprawdź `du -sh assets/`.
- **Tylko assety CC0.** Żadnych z Unity Asset Store — licencja zabrania redystrybucji, a repo jest publiczne. Każdy pobrany plik musi trafić do `assets/museum/LICENSES.md`.
- **Cache-busting:** po każdej zmianie w `js/` lub `css/` podbij `?v=` w `museum.html` na `date '+%Y%m%d%H%M'`.
- **Język interfejsu: polski**, z pełną ortografią (ą, ć, ę, ł, ń, ó, ś, ź, ż). Komentarze w kodzie po polsku, zgodnie z resztą repo.
- **Nie ruszaj `index.html`, `js/main.js`, `css/main.css` ani `js/projects-data.js`.** Muzeum czyta dane, nie zmienia ich.
- **Nie dodawaj rachunku tokenów z powrotem** — został usunięty decyzją użytkownika (`d221f65`).

- [ ] **Krok 4: Napisz `js/museum/textures.js`**

```js
/* Ładowanie materiałów PBR. Jedna mapa ARM obsługuje trzy sloty naraz:
   R = ambient occlusion, G = roughness, B = metalness. */
import * as THREE from "three";

const loader = new THREE.TextureLoader();
const cache = new Map();

function wczytaj(sciezka, powtorzenia, srgb) {
  const t = loader.load(sciezka);
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  t.repeat.set(powtorzenia, powtorzenia);
  t.anisotropy = 8;
  if (srgb) t.colorSpace = THREE.SRGBColorSpace;
  return t;
}

export function loadPBR(nazwa, powtorzenia = 4) {
  const klucz = `${nazwa}@${powtorzenia}`;
  if (cache.has(klucz)) return cache.get(klucz);
  const b = `assets/museum/${nazwa}`;
  const arm = wczytaj(`${b}_arm.webp`, powtorzenia, false);
  const mat = new THREE.MeshStandardMaterial({
    map: wczytaj(`${b}_kolor.webp`, powtorzenia, true),
    normalMap: wczytaj(`${b}_normal.webp`, powtorzenia, false),
    aoMap: arm, roughnessMap: arm, metalnessMap: arm,
    roughness: 1, metalness: 1,   // mnożniki — prawdziwe wartości są w mapie ARM
  });
  cache.set(klucz, mat);
  return mat;
}
```

`aoMap` wymaga drugiego zestawu UV (`uv1`). Dla `PlaneGeometry` i `BoxGeometry` skopiuj kanał: `geo.setAttribute("uv1", geo.attributes.uv);` — bez tego AO jest po cichu ignorowane.

```js
export function buildBuilding(dlugosciSal) {
  const matBeton = loadPBR("beton", 8), matParkiet = loadPBR("parkiet", 6), matTynk = loadPBR("tynk", 4);
  // UWAGA (ustalenie z Zadania 3): loadPBR bierze JEDEN skalar powtórzeń na obie osie,
  // więc na powierzchni niekwadratowej kafle wychodzą rozciągnięte. Sala 14×18 jeszcze
  // to wybaczy, ale ściana 5 m wysoka na 20 m długa już nie. Materiały są współdzielone
  // przez cache, więc NIE modyfikuj `.map.repeat` na gotowym materiale — popsujesz
  // wszystkie powierzchnie, które go używają. Jeśli kafle wyjdą wyraźnie rozciągnięte,
  // sklonuj materiał (`mat.clone()` plus klon tekstur) albo poszerz loadPBR o drugi
  // parametr osi — i zgłoś to jako odstępstwo w raporcie.
  const kolizje = new THREE.Group();
  const sale = [];
  const root = new THREE.Group();
  root.add(atrium(matBeton, matTynk));

  let z = SZER_ATRIUM / 2;
  dlugosciSal.forEach((dl, i) => {
    const odZ = z;
    const sala = new THREE.Group();
    sala.position.z = z;
    // podłoga
    const pod = new THREE.Mesh(new THREE.PlaneGeometry(SZER_SALI, dl), matParkiet);
    pod.geometry.setAttribute("uv1", pod.geometry.attributes.uv);
    pod.rotation.x = -Math.PI / 2; pod.position.z = dl / 2; pod.receiveShadow = true;
    sala.add(pod);
    // ściany boczne
    [-1, 1].forEach((s) => {
      const w = bx(0.4, WYS_SALI, dl, matTynk);
      w.position.set(s * SZER_SALI / 2, WYS_SALI / 2, dl / 2);
      w.castShadow = w.receiveShadow = true;
      sala.add(w);
    });
    // sufit
    const suf = bx(SZER_SALI, 0.4, dl, matTynk);
    suf.position.set(0, WYS_SALI, dl / 2);
    sala.add(suf);
    // ściana tylna z portalem (ostatnia sala dostaje pełną)
    const ostatnia = i === dlugosciSal.length - 1;
    const tyl = ostatnia
      ? bx(SZER_SALI, WYS_SALI, 0.4, matTynk)
      : scianaZDziura(SZER_SALI, WYS_SALI, 0.4, SZER_PORTALU, WYS_PORTALU, matTynk);
    if (ostatnia) tyl.position.y = WYS_SALI / 2;
    tyl.position.z = dl;
    sala.add(tyl);

    root.add(sala);
    sale.push({ id: i + 1, odZ, doZ: z + dl, srodekZ: z + dl / 2 });
    z += dl;
  });

  scene.add(root);
  // Warstwa kolizyjna: te same bryły, bez materiałów i bez eksponatów.
  // POPRAWKA po przeglądzie Zadania 4: rozpoznawanie po `geometry.type === "BoxGeometry"`
  // było błędem — pomijało siedem podłóg (to PlaneGeometry), więc Zadanie 5 z grawitacją
  // przepuściłoby gracza przez podłogę w pierwszej klatce. Do tego wciągało emisyjne
  // listwy epok, które też są BoxGeometry, i gracz kleiłby się do nich przy ścianie.
  // Rozpoznajemy więc po jawnym znaczniku `userData.kolizja`, a pod każdą podłogę
  // podkładamy cienką bryłę — Octree potrzebuje objętości, nie płaszczyzny.
  root.traverse((o) => {
    if (!o.isMesh || !o.userData.kolizja) return;
    const k = new THREE.Mesh(o.geometry);
    o.getWorldPosition(k.position);
    o.getWorldQuaternion(k.quaternion);
    o.getWorldScale(k.scale);        // dziś wszędzie 1, ale bez tego skala rozjedzie
    kolizje.add(k);                  // kolizje z obrazem bez żadnego sygnału
  });
  return { kolizje, sale };
}
```

```js
/* Chodzenie z kolizjami. Octree + Capsule to rozwiązanie z oficjalnego dema FPS
   three.js — daje ślizganie po ścianach i wchodzenie w portale za darmo, czego
   ręczne AABB nie potrafi (zakleszcza się w narożnikach). */
import * as THREE from "three";
import { PointerLockControls } from "three/addons/controls/PointerLockControls.js";
import { Octree } from "three/addons/math/Octree.js";
import { Capsule } from "three/addons/math/Capsule.js";
import { camera, renderer, reduceMotion } from "./render.js";

const WZROST = 1.7, PROMIEN = 0.35, PREDKOSC = 5.2, BIEG = 1.9, GRAWITACJA = 22;

export function initPlayer(kolizje) {
  const octree = new Octree().fromGraphNode(kolizje);
  const kapsula = new Capsule(
    new THREE.Vector3(0, PROMIEN, 0),
    new THREE.Vector3(0, WZROST, 0),
    PROMIEN
  );
  const controls = new PointerLockControls(camera, renderer.domElement);
  const klawisze = {};
  const predkosc = new THREE.Vector3();
  let naZiemi = false;

  addEventListener("keydown", (e) => { klawisze[e.code] = true; });
  addEventListener("keyup", (e) => { klawisze[e.code] = false; });
  renderer.domElement.addEventListener("click", () => controls.lock());

  function kierunek() {
    const przod = new THREE.Vector3();
    camera.getWorldDirection(przod);
    przod.y = 0; przod.normalize();
    const bok = new THREE.Vector3().crossVectors(przod, camera.up).normalize();
    const d = new THREE.Vector3();
    if (klawisze.KeyW || klawisze.ArrowUp) d.add(przod);
    if (klawisze.KeyS || klawisze.ArrowDown) d.sub(przod);
    if (klawisze.KeyD || klawisze.ArrowRight) d.add(bok);
    if (klawisze.KeyA || klawisze.ArrowLeft) d.sub(bok);
    return d.normalize();
  }

  function kolizja() {
    const w = octree.capsuleIntersect(kapsula);
    naZiemi = false;
    if (!w) return;
    naZiemi = w.normal.y > 0;
    if (!naZiemi) predkosc.addScaledVector(w.normal, -w.normal.dot(predkosc));
    kapsula.translate(w.normal.multiplyScalar(w.depth));
  }

  return {
    controls,
    zablokowany: () => controls.isLocked,
    pozycja: () => kapsula.end.clone(),
    teleportuj(z) {
      kapsula.start.set(0, PROMIEN, z);
      kapsula.end.set(0, WZROST, z);
    },
    update(dt) {
      const tempo = PREDKOSC * (klawisze.ShiftLeft ? BIEG : 1);
      const d = kierunek();
      predkosc.x = d.x * tempo; predkosc.z = d.z * tempo;
      if (!naZiemi) predkosc.y -= GRAWITACJA * dt; else predkosc.y = 0;
      kapsula.translate(predkosc.clone().multiplyScalar(dt));
      kolizja();
      camera.position.copy(kapsula.end);
      // bujanie kroku — wyłączone przy prefers-reduced-motion
      if (!reduceMotion && d.lengthSq() > 0 && naZiemi) {
        camera.position.y += Math.sin(performance.now() * 0.011 * tempo) * 0.025;
      }
    },
  };
}
```

```js
/* Degradacja jednokierunkowa: raz wyłączonego efektu nie włączamy z powrotem,
   bo na granicy wydajności scena migotałaby w tę i z powrotem. */
export function initPerf({ composer, bloom, renderer, komunikat }) {
  let klatki = 0, suma = 0, stopien = 0;
  return function tick(dt) {
    if (stopien >= 2) return;
    suma += dt; klatki++;
    if (klatki < 90) return;
    const fps = klatki / suma;
    klatki = 0; suma = 0;
    if (fps >= 25) return;
    if (stopien === 0) {
      composer.removePass(bloom);
      komunikat("Wyłączyłem poświatę, żeby złapać płynność.");
    } else {
      renderer.shadowMap.enabled = false;
      renderer.shadowMap.needsUpdate = true;
      komunikat("Wyłączyłem też cienie — ten sprzęt nie wyrabia.");
    }
    stopien++;
  };
}
```
